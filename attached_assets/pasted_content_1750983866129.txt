export async function sendVerificationCode(email: string) {
  const code = Math.floor(100000 + Math.random() * 900000).toString()
  // upsert into `verification` table
  await prisma.verification.upsert({
    where: { email },
    update: { code, expiresAt: new Date(Date.now() + 15 * 60e3) },
    create: { email, code, expiresAt: new Date(Date.now() + 15 * 60e3) }
  })
  // send email
  const transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port: +process.env.SMTP_PORT!,
    secure: true,
    auth: {
      user: process.env.SMTP_USER,
      pass: process.env.SMTP_PASS
    }
  })
  await transporter.sendMail({
    from: `"Harmony” <${process.env.SMTP_FROM}>`,
    to: email,
    subject: 'Your Harmony verification code',
    text: `Your verification code is ${code}. It expires in 15 minutes.`
  })
}

// --- 2. Verify code & activate user ---
export async function checkVerificationCode(email: string, code: string) {
  const record = await prisma.verification.findUnique({ where: { email } })
  if (!record) return false
  if (record.code !== code) return false
  if (record.expiresAt < new Date()) return false
  return true
}

export async function activateUser(email: string) {
  // set status active and remove verification record
  await prisma.user.update({
    where: { email },
    data: { status: 'active' }
  })
  await prisma.verification.delete({ where: { email } })
}

import NextAuth from 'next-auth'
import CredentialsProvider from 'next-auth/providers/credentials'
import { PrismaClient } from '@prisma/client'
import bcrypt from 'bcrypt'

const prisma = new PrismaClient()

export default NextAuth({
  providers: [
    CredentialsProvider({
      name: 'Email / Password',
      credentials: {
        email: { label: 'Email', type: 'text' },
        password: { label: 'Password', type: 'password' }
      },
      async authorize(credentials) {
        const user = await prisma.user.findUnique({
          where: { email: credentials.email }
        })
        if (user && user.status === 'active') {
          const valid = await bcrypt.compare(credentials.password, user.password)
          if (valid) {
            return { id: user.id, email: user.email }
          }
        }
        return null
      }
    })
  ],
  session: { strategy: 'jwt' },
  callbacks: {
    async jwt({ token, user }) {
      if (user) token.userId = (user as any).id
      return token
    },
    async session({ session, token }) {
      if (token) session.user = { id: token.userId, email: session.user.email }
      return session
    }
  },
  pages: {
    signIn: '/auth/login',
    error: '/auth/login'
  }
})

3. /components/builder/store.ts (with undo/redo)

import create from 'zustand'
import produce from 'immer'
import type { Node, Edge } from 'react-flow-renderer'

type State = {
  nodes: Node[]
  edges: Edge[]
  history: { nodes: Node[]; edges: Edge[] }[]
  future: { nodes: Node[]; edges: Edge[] }[]
  setNodes: (fn: (curr: Node[]) => Node[]) => void
  setEdges: (fn: (curr: Edge[]) => Edge[]) => void
  undo: () => void
  redo: () => void
  exportFlow: () => void
  importFlow: (json: string) => void
}

const useStore = create<State>((set, get) => ({
  nodes: [],
  edges: [],
  history: [],
  future: [],

  setNodes: fn => {
    set(state => {
      const next = fn(state.nodes)
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.future = []
      state.nodes = next
    })
  },

  setEdges: fn => {
    set(state => {
      const next = fn(state.edges)
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.future = []
      state.edges = next
    })
  },

  undo: () => {
    set(state => {
      if (state.history.length === 0) return
      const prev = state.history.pop()!
      state.future.push({ nodes: state.nodes, edges: state.edges })
      state.nodes = prev.nodes
      state.edges = prev.edges
    })
  },

  redo: () => {
    set(state => {
      if (state.future.length === 0) return
      const next = state.future.pop()!
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.nodes = next.nodes
      state.edges = next.edges
    })
  },

  exportFlow: () => {
    const data = JSON.stringify({ nodes: get().nodes, edges: get().edges }, null, 2)
    const blob = new Blob([data], { type: 'application/json' })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'flow.json'
    a.click()
  },

  importFlow: (json) => {
    try {
      const { nodes, edges } = JSON.parse(json)
      set(() => ({ nodes, edges, history: [], future: [] }))
    } catch (e) { console.error('Invalid JSON') }
  }
}))

export default useStore

import { NextApiRequest, NextApiResponse } from 'next'
import { authGuard } from '@/lib/middleware/security'
import { PrismaClient } from '@prisma/client'
import fetch from 'node-fetch'

const prisma = new PrismaClient()
const handler = nc<NextApiRequest, NextApiResponse>()

handler.use(authGuard)

handler.post(async (req, res) => {
  const { prompt } = req.body
  const userId = (req as any).user.id
  if (!prompt) return res.status(400).json({ message: 'Prompt is required' })

  // Call Orchestrator to scaffold agent config
  const orchestratorRes = await fetch(`${process.env.APP_URL}/api/orchestrator`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      Authorization: req.headers.authorization!
    },
    body: JSON.stringify({ taskType: 'createAgent', clientData: { userId, prompt } })
  })
  if (!orchestratorRes.ok) {
    const err = await orchestratorRes.text()
    return res.status(500).json({ message: 'Orchestrator error', details: err })
  }
  const { result } = await orchestratorRes.json()

  // Persist new Agent
  const agent = await prisma.agent.create({
    data: {
      userId,
      prompt,
      config: result,
      status: 'active'
    }
  })
  res.status(200).json(agent)
})

5.1 Store with Selection ( /components/builder/store.ts )
import create from 'zustand'
import type { Node, Edge } from 'react-flow-renderer'

type State = {
  nodes: Node[]
  edges: Edge[]
  selectedNodeId: string | null
  history: { nodes: Node[]; edges: Edge[] }[]
  future: { nodes: Node[]; edges: Edge[] }[]
  setNodes: (fn: (curr: Node[]) => Node[]) => void
  setEdges: (fn: (curr: Edge[]) => Edge[]) => void
  setSelectedNode: (id: string | null) => void
  undo: () => void
  redo: () => void
  exportFlow: () => void
  importFlow: (json: string) => void
}

const useStore = create<State>((set, get) => ({
  nodes: [],
  edges: [],
  selectedNodeId: null,
  history: [],
  future: [],

  setNodes: fn => {
    set(state => {
      const next = fn(state.nodes)
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.future = []
      state.nodes = next
    })
  },

  setEdges: fn => {
    set(state => {
      const next = fn(state.edges)
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.future = []
      state.edges = next
    })
  },

  setSelectedNode: id => set({ selectedNodeId: id }),

  undo: () => {
    set(state => {
      if (!state.history.length) return
      const prev = state.history.pop()!
      state.future.push({ nodes: state.nodes, edges: state.edges })
      state.nodes = prev.nodes
      state.edges = prev.edges
    })
  },

  redo: () => {
    set(state => {
      if (!state.future.length) return
      const next = state.future.pop()!
      state.history.push({ nodes: state.nodes, edges: state.edges })
      state.nodes = next.nodes
      state.edges = next.edges
    })
  },

  exportFlow: () => {
    const data = JSON.stringify({ nodes: get().nodes, edges: get().edges }, null, 2)
    const blob = new Blob([data], { type: 'application/json' })
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = 'flow.json'
    a.click()
  },

  importFlow: json => {
    try {
      const { nodes, edges } = JSON.parse(json)
      set(() => ({ nodes, edges, history: [], future: [], selectedNodeId: null }))
    } catch { console.error('Invalid JSON') }
  }
}))

export default useStore

import React, { useState, useEffect } from 'react'
import useStore from './store'

export default function PropertiesPanel() {
  const selectedId = useStore(s => s.selectedNodeId)
  const nodes      = useStore(s => s.nodes)
  const setNodes   = useStore(s => s.setNodes)
  const [label, setLabel] = useState('')
  const [desc, setDesc]   = useState('')

  useEffect(() => {
    if (selectedId) {
      const node = nodes.find(n => n.id === selectedId)
      setLabel(node?.data?.label || '')
      setDesc(node?.data?.description || '')
    }
  }, [selectedId, nodes])

  if (!selectedId) {
    return (
      <aside className="properties-panel p-4 border-l">
        <p className="italic">Select a node to edit its properties</p>
      </aside>
    )
  }

  const save = () => {
    setNodes(curr =>
      curr.map(n =>
        n.id === selectedId
          ? { ...n, data: { ...n.data, label, description: desc } }
          : n
      )
    )
  }

  return (
    <aside className="properties-panel p-4 border-l w-64">
      <h3 className="text-lg mb-2">Node Properties</h3>
      <label className="block mb-1">Label</label>
      <input
        value={label}
        onChange={e => setLabel(e.target.value)}
        className="w-full border p-1 mb-3 rounded"
      />
      <label className="block mb-1">Description</label>
      <textarea
        value={desc}
        onChange={e => setDesc(e.target.value)}
        className="w-full border p-1 mb-3 rounded"
      />
      <button onClick={save} className="btn w-full">
        Save
      </button>

  )
}

import { NextApiRequest, NextApiResponse } from 'next'
import Stripe from 'stripe'

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: '2022-11-15' })

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  if (req.method !== 'POST') return res.status(405).end()
  const { planId, oneTime } = req.body

  try {
    const sessionParams: Stripe.Checkout.SessionCreateParams = {
      payment_method_types: ['card'],
      success_url: `${process.env.APP_URL}/auth/onboarding`,
      cancel_url: `${process.env.APP_URL}/auth/plans`,
    }

    if (oneTime) {
      // one-time purchase
      sessionParams.mode = 'payment'
      sessionParams.line_items = [{ price: planId, quantity: 1 }]
    } else {
      // subscription
      sessionParams.mode = 'subscription'
      sessionParams.line_items = [{ price: planId, quantity: 1 }]
    }

    const session = await stripe.checkout.sessions.create(sessionParams)
    return res.status(200).json({ sessionId: session.id, url: session.url })
  } catch (err: any) {
    return res.status(400).json({ message: err.message })
  }
}

// pages/api/profiles/[username].ts
import { NextApiRequest, NextApiResponse } from 'next'
import { PrismaClient } from '@prisma/client'
const prisma = new PrismaClient()

export default async function handler(req: NextApiRequest, res: NextApiResponse) {
  const { username } = req.query
  const user = await prisma.user.findUnique({
    where: { username: String(username) },
    select: {
      id: true, name: true, bio: true, avatarUrl: true, type: true,
      services: true,
      agents: { select: { id: true, slug: true, name: true, price: true, imageUrl: true } }
    }
  })
  if (!user) return res.status(404).end()
  res.json(user)

2. Stripe Webhook Handler
File: /pages/api/webhooks/stripe.ts

import { buffer } from 'micro'
import Stripe from 'stripe'
import { PrismaClient } from '@prisma/client'

export const config = { api: { bodyParser: false } }
const prisma = new PrismaClient()
const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: '2022-11-15' })

export default async function handler(req, res) {
  const sig = req.headers['stripe-signature']
  const buf = await buffer(req)
  let event: Stripe.Event
  try {
    event = stripe.webhooks.constructEvent(buf, sig, process.env.STRIPE_WEBHOOK_SECRET!)
  } catch (err) {
    return res.status(400).send(`Webhook Error: ${err.message}`)
  }

  switch (event.type) {
    case 'invoice.payment_succeeded':
      // update subscription status
      const invoice = event.data.object as Stripe.Invoice
      await prisma.subscription.update({
        where: { stripeSubscriptionId: invoice.subscription as string },
        data: { status: 'active' }
      })
      break
    case 'invoice.payment_failed':
      const failed = event.data.object as Stripe.Invoice
      await prisma.subscription.update({
        where: { stripeSubscriptionId: failed.subscription as string },
        data: { status: 'past_due' }
      })
      // notify user...
      break
    case 'customer.subscription.deleted':
      const sub = event.data.object as Stripe.Subscription
      await prisma.subscription.update({
        where: { stripeSubscriptionId: sub.id },
        data: { status: 'canceled' }
      })
      break
    // add other events as needed
    default:
      console.log(`Unhandled event type ${event.type}`)
  }

  res.json({ received: true })
}
