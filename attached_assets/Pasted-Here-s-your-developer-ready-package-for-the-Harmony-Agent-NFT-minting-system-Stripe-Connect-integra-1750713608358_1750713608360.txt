🔐 HarmonyAgentNFT Smart Contract (Solidity)

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.17;

import "@openzeppelin/contracts/token/ERC721/extensions/ERC721URIStorage.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

contract HarmonyAgentNFT is ERC721URIStorage, Ownable {
    uint256 public nextTokenId;
    address public marketplace;
    mapping(uint256 => uint256) public agentRoyalties; // agentId => royalty %

    constructor(address _marketplace) ERC721("HarmonyAgent", "HAGENT") {
        marketplace = _marketplace;
    }

    function mintAgent(address to, string memory tokenURI, uint256 royalty) public onlyOwner {
        require(royalty <= 1000, "Royalty too high"); // Max 10%
        uint256 newTokenId = nextTokenId++;
        _safeMint(to, newTokenId);
        _setTokenURI(newTokenId, tokenURI);
        agentRoyalties[newTokenId] = royalty;
    }

    function updateMarketplace(address _new) public onlyOwner {
        marketplace = _new;
    }

    function getRoyalty(uint256 tokenId) public view returns (uint256) {
        return agentRoyalties[tokenId];
    }
}

🔁 Stripe Connect Button (JavaScript)

// JS (React-like pseudocode): Stripe Connect Onboarding UI
import { useEffect } from "react";

function StripeConnectButton({ userId }) {
    const handleConnect = async () => {
        const res = await fetch('/api/stripe/create-connect-link', {
            method: 'POST',
            body: JSON.stringify({ userId }),
            headers: { 'Content-Type': 'application/json' },
        });
        const { url } = await res.json();
        window.location.href = url;
    };

    return <button onClick={handleConnect}>Connect Stripe Account</button>;
}
