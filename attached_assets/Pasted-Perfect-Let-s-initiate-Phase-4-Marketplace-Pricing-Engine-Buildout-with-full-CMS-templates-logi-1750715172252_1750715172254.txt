⚙️ LOGIC FLOW:

function calculateSetupFee(plan) {
  const plans = {
    starter: 4400,
    growth: 8000,
    expansion: 15000,
    enterprise: 21500
  }
  return plans[plan]
}

function calculateAgentMonthly(agentCount) {
  const base = 444
  if (agentCount >= 6) return base * agentCount * 0.85
  if (agentCount >= 3) return base * agentCount * 0.9
  return base * agentCount
}

function showSetupFee(user) {
  return user.hasPlatformInstalled ? false : true
}
