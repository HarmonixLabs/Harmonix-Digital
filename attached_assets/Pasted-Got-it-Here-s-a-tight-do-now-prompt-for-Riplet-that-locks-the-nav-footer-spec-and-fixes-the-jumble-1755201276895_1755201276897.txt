// components/nav/Navbar.tsx
import Link from "next/link";
import { useState, useEffect } from "react";
import clsx from "clsx";

export default function Navbar({ solStatus = "disconnected", isAuthed = false }:{
  solStatus?: "connected"|"disconnected",
  isAuthed?: boolean
}) {
  const [atTop, setAtTop] = useState(true);
  useEffect(() => {
    const onScroll = () => setAtTop(window.scrollY < 8);
    onScroll(); window.addEventListener("scroll", onScroll);
    return () => window.removeEventListener("scroll", onScroll);
  }, []);

  const Dot = ({ ok }: { ok: boolean }) => (
    <span className={clsx("inline-block h-2 w-2 rounded-full mr-2",
      ok ? "bg-[#16a34a]" : "bg-[#ef4444]")} />
  );

  return (
    <header className={clsx(
      "sticky top-0 z-50 backdrop-blur supports-[backdrop-filter]:bg-[#0b0b0bcc] bg-[#0b0b0c]/85",
      atTop ? "border-b border-transparent" : "border-b border-white/5"
    )}>
      <nav className="mx-auto max-w-7xl px-4 lg:px-6">
        <div className="flex h-14 items-center justify-between">
          {/* Left: Logo */}
          <Link href="/" className="flex items-center gap-2">
            <img src="/brand/monogram.svg" alt="Harmonix" className="h-6 w-6" />
            <span className="sr-only">Harmonix</span>
          </Link>

          {/* Center: Primary Nav (desktop) */}
          <ul className="hidden lg:flex items-center gap-6 text-sm text-white/80">
            <li><Link className="hover:text-white" href="/">Home</Link></li>
            <li><Link className="hover:text-white" href="/marketplace">Marketplace</Link></li>
            <li><Link className="hover:text-white" href="/demo">Demo</Link></li>
            <li><Link className="hover:text-white" href="/pricing">Pricing</Link></li>
            <li>
              <Link
                href="/join"
                className="rounded-full px-3 py-1.5 text-sm font-medium bg-[#FDC500] text-black hover:shadow-[0_0_0_3px_rgba(253,197,0,.25)] transition">
                Join – Early Access
              </Link>
            </li>
            <li>
              <Link className="hover:text-white" href={isAuthed ? "/dashboard" : "/auth/signin"}>
                {isAuthed ? "Dashboard" : "Sign In"}
              </Link>
            </li>
          </ul>

          {/* Right: SOL chip (desktop) */}
          <div className="hidden lg:flex items-center text-xs text-white/80">
            <span className="inline-flex items-center rounded-full border border-white/10 px-2 py-1">
              <Dot ok={solStatus === "connected"} />
              {solStatus === "connected" ? "SŌL Connected" : "SŌL Disconnected"}
            </span>
          </div>

          {/* Mobile burger */}
          <MobileMenu isAuthed={isAuthed} solStatus={solStatus} />
        </div>
      </nav>
    </header>
  );
}

function MobileMenu({ isAuthed, solStatus }:{ isAuthed:boolean, solStatus:"connected"|"disconnected" }) {
  const [open,setOpen]=useState(false);
  return (
    <div className="lg:hidden">
      <button aria-label="Open menu" onClick={()=>setOpen(true)} className="p-2 text-white/80">☰</button>
      {open && (
        <div className="fixed inset-0 z-50 bg-[#0b0b0f]/95 backdrop-blur">
          <div className="mx-auto max-w-lg p-6 space-y-4">
            <button onClick={()=>setOpen(false)} className="text-white/70">Close ✕</button>
            <nav className="flex flex-col text-lg text-white/90 gap-3">
              <Link href="/" onClick={()=>setOpen(false)}>Home</Link>
              <Link href="/marketplace" onClick={()=>setOpen(false)}>Marketplace</Link>
              <Link href="/demo" onClick={()=>setOpen(false)}>Demo</Link>
              <Link href="/pricing" onClick={()=>setOpen(false)}>Pricing</Link>
              <Link
                href="/join"
                onClick={()=>setOpen(false)}
                className="mt-2 rounded-xl px-4 py-3 bg-[#FDC500] text-black font-semibold">
                Join – Early Access
              </Link>
              <Link href={isAuthed?"/dashboard":"/auth/signin"} onClick={()=>setOpen(false)}>
                {isAuthed? "Dashboard":"Sign In"}
              </Link>
              <div className="pt-4 text-sm text-white/70">
                <span className="inline-flex items-center gap-2">
                  <span className={`h-2 w-2 rounded-full ${solStatus==="connected"?"bg-[#16a34a]":"bg-[#ef4444]"}`}/>
                  {solStatus==="connected"?"SŌL Connected":"SŌL Disconnected"}
                </span>
              </div>
            </nav>
          </div>
        </div>
      )}
    </div>
  );
}
